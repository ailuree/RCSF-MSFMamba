# A-MyCode 常用指令

## 1. 进入 WSL 工作目录
cd /home/project/5-MyProject/A-MyCode


## 2. 基线正式实验

# Houston2013 基线正式训练
python train.py --dataset Houston2013 --batchsize 8 --epoch 40 --num_work 4 --run_name baseline_h2013

# Houston2013 基线独立评估
python eval.py --dataset Houston2013 --run_name baseline_h2013

# Houston2013 生成预测图 / GT 图
python visualize.py --dataset Houston2013 --run_name baseline_h2013 --split all

# Houston2013 生成预测图 / GT 图 / HSI RGB / LiDAR / 四联图
python visualize.py --dataset Houston2013 --run_name baseline_h2013 --split all --save_input_views 1

# Houston2018 基线正式训练（开发期建议 small）
python train.py --dataset Houston2018 --batchsize 8 --epoch 40 --num_work 4 --eval_split small --run_name baseline_h2018

# Houston2018 基线独立评估
python eval.py --dataset Houston2018 --run_name baseline_h2018 --eval_split small

# Houston2018 生成预测图 / GT 图 / 四联图
python visualize.py --dataset Houston2018 --run_name baseline_h2018 --split all --eval_split small --save_input_views 1


## 3. 开发期 smoke test

# Houston2013 快速 smoke test：少量训练 batch，跳过测试
python train.py --dataset Houston2013 --batchsize 4 --epoch 1 --num_work 2 --skip_test 1 --max_train_batches 20 --print_freq 5 --run_name smoke_h2013

# Houston2013 快速 smoke test：少量训练 + 少量测试
python train.py --dataset Houston2013 --batchsize 4 --epoch 1 --num_work 2 --max_train_batches 20 --max_test_batches 10 --print_freq 5 --run_name smoke_h2013_eval

# Houston2018 快速 smoke test：少量训练 + 少量测试
python train.py --dataset Houston2018 --batchsize 4 --epoch 1 --num_work 2 --eval_split small --max_train_batches 20 --max_test_batches 10 --print_freq 5 --run_name smoke_h2018


## 4. 常用调参方式

# 提高 batchsize（先观察显存）
python train.py --dataset Houston2013 --batchsize 12 --epoch 40 --num_work 4 --run_name baseline_h2013_bs12

# 提高 dataloader worker
python train.py --dataset Houston2013 --batchsize 8 --epoch 40 --num_work 6 --run_name baseline_h2013_w6

# 调整打印频率
python train.py --dataset Houston2013 --batchsize 8 --epoch 40 --num_work 4 --print_freq 50 --run_name baseline_h2013_pf50


## 5. 只做独立评估 / 可视化

# 指定某个 checkpoint 做评估
python eval.py --dataset Houston2013 --checkpoint ./checkpoints/Houston2013/baseline_h2013/weights/best.pth

# 指定某个 checkpoint 生成图
python visualize.py --dataset Houston2013 --checkpoint ./checkpoints/Houston2013/baseline_h2013/weights/best.pth --split all --save_input_views 1
