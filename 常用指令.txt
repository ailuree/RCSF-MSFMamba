# A-MyCode 常用指令

## 1. 进入 WSL 工作目录
cd /home/project/5-MyProject/A-MyCode


## 2. Houston2018 常用训练/验证

# 开发期推荐：默认小测试集验证
python train.py --dataset Houston2018 --batchsize 8 --epoch 1 --num_work 4 --eval_split small

# Houston2018 完整测试集验证
python train.py --dataset Houston2018 --batchsize 8 --epoch 1 --num_work 4 --eval_split full

# Houston2018 快速 smoke test：少量训练 batch，跳过测试
python train.py --dataset Houston2018 --batchsize 4 --epoch 1 --num_work 2 --skip_test 1 --max_train_batches 20 --print_freq 5

# Houston2018 快速 smoke test：少量训练 + 少量测试
python train.py --dataset Houston2018 --batchsize 4 --epoch 1 --num_work 2 --max_train_batches 20 --max_test_batches 10 --print_freq 5


## 3. Houston2013 常用训练/验证

# Houston2013 完整跑通 1 个 epoch
python train.py --dataset Houston2013 --batchsize 8 --epoch 1 --num_work 2

# Houston2013 快速 smoke test
python train.py --dataset Houston2013 --batchsize 4 --epoch 1 --num_work 2 --skip_test 1 --max_train_batches 20 --print_freq 5


## 4. 其他数据集示例

# Berlin
python train.py --dataset Berlin --batchsize 8 --epoch 1 --num_work 2

# Augsburg
python train.py --dataset Augsburg --batchsize 8 --epoch 1 --num_work 2


## 5. 常用调参方式

# 提高 batchsize（先观察显存）
python train.py --dataset Houston2018 --batchsize 12 --epoch 1 --num_work 4 --eval_split small

# 提高 dataloader worker
python train.py --dataset Houston2018 --batchsize 8 --epoch 1 --num_work 6 --eval_split small

# 调整打印频率
python train.py --dataset Houston2018 --batchsize 8 --epoch 1 --num_work 4 --eval_split small --print_freq 50
